\section{Un peu d'histoire}
We start with some motivation: this course will be about number fields (i.e. finite extensions $K|\Q$) and their absolute Galois groups $G_K = \Gal(\overline{K}|K)$, where $\overline{K}$ is a fixed algebraic closure of $K$. Very little is known about the structure of such automorphism groups in general, and a large chunk of modern number theory/arithmetic geometry is devoted to studying them, e.g. via their representations. In some sense, class field theory consists in the study of its $1$-dimensional representations; more precisely, it describes the abelianization $G_K^{\ab} = G_K/[G_K,G_K]$ of the absolute Galois group (equivalently, of its abelian quotients).

\subsection{Reciprocity laws}
Let $p$ be a prime number, and let $a$ be an integer coprime to $p$. The \emph{Legendre symbol} $\legendre{a}{p}$ is defined as
\[
	\legendre{a}{p} = \begin{cases}
		1  & \text{if $a$ is a square modulo $p$},     \\
		-1 & \text{if $a$ is not a square modulo $p$}.
	\end{cases}
\]
\textbf{Euler} proved that $\legendre{a}{p} = a^{\frac{p-1}{2}} \bmod p$, and \textbf{Gauss} proved the following law:
\begin{thm}[Quadratic reciprocity]
	Let $a\ne b$ be different odd primes. Then,
	\[
		\legendre {a}{b}\cdot\legendre{b}{a} = (-1)^{\frac{a-1}{2}\frac{b-1}{2}}.
	\]
	Morever, the following subsidiary laws hold:
	\begin{enumerate}
		\item $\legendre{-1}{p} = (-1)^{\frac{p-1}{2}}$;
		\item $\legendre{2}{p} = (-1)^{\frac{p^2-1}{8}}$.
	\end{enumerate}
\end{thm}
These laws, together with its multiplicativity in the parameter $a$, make it possible to compute the Legendre symbol.

Another way of thinking about class field theory is as an attempt to define a ``very general'' reciprocity law, from which the quadratic reciprocity law and its higher degree analogues can be deduced. One such generalization is \textbf{Hilbert}'s reciprocity law: recall that $\Q$ embeds into the $p$-adic field $\Q_p$ for every prime $p$, and also into the real numbers $\Q_\infty\coloneqq\R$.
For every prime $p$ (including $p=\infty$) and for all $a,b\in \Q^\times$, we define the \emph{Hilbert symbol} $(a,b)_p$ as
\[
	(a,b)_p = \begin{cases}
		1  & \text{if $ax^2 + by^2 = 1$ has a solution in $\Q_p^2$}, \\
		-1 & \text{otherwise}.
	\end{cases} asd
\]

Recall that the norm $N:\Q_p(\sqrt{a})\to \Q_p$ is defined as $N(x+y\sqrt{a}) = x^2 - ay^2$ if $a\notin \Q_p^{\times 2}$, and as $N=\id$ otherwise. More generally, if $L|K$ is a finite Galois extension with group $G$, the norm $N_{L|K}:L\to K$ is
\[
	N(x)=\prod_{\sigma\in G} \sigma(x)
\]
\begin{lem}
	The following statements are equivalent:
	\begin{enumerate}
		\item $(a,b)_p = 1$;
		\item The quadratic equation $z^2 = ax^2 + by^2$ has a non-trivial solution;
		\item $a\in \Q_p$ is the norm of an element of $\Q_p(\sqrt{b})$.
		\item $b\in \Q_p$ is the norm of an element of $\Q_p(\sqrt{a})$.
	\end{enumerate}
\end{lem}
\begin{proof}
	The equivalence $1\iff 2$ is by definition; and $3\iff 4$ is clear.
	For $2\iff 3$, if $b\in \Q_p^{\times 2}$, then we are done. Otherwise, if $b\notin \Q_p^{\times 2}$, then $N(\alpha+\beta\sqrt{b})= \alpha^2 - b\beta^2$, and if $a\coloneqq \alpha^2 - b\beta^2$ we see that $z^2 = ax^2 + by^2$ has a non-trivial solution, and vice versa.
\end{proof}
For $p\ne 2, \infty$ and $a,b\in\Z$, write
$a=p^{v_p(a)}u$ and $b=p^{v_p(b)}v$ with $p\nmid uv$. The relation between the Hilbert symbol and the Legendre symbol is
\begin{align}
	(a,b)_p
	 & = (-1)^{\frac{p-1}{2}v_p(a)v_p(b)}
	\legendre{u}{p}^{v_p(b)}\legendre{v}{p}^{v_p(a)}                    \\
	 & = \overline{(-1)^{v_p(a)v_p(b)}a^{v_p(b)}b^{-v_p(a)}}^{(p-1)/2}.
\end{align}
where the bar denotes reduction modulo $p$.
In particular, if $p$ does not divide $a$, then $(a,b)_p = \legendre{a}{p}^{v_p(b)}$ coincides with the Legendre symbol.
\begin{thm}[Hilbert reciprocity law]
	For all $a,b\in \Q^\times$, we have
	\[
		\prod_{p} (a,b)_p = 1
	\]
\end{thm}
As for computation, one shows that if $p$ does not divide $2ab$ (including $p=\infty$ for $a,b>0$), then $(a,b)_p = 1$. Moreover, $(a,b)_2=(-1)^{\frac{a-1}{2}\frac{b-1}{2}}$ for odd $a,b$. From this and multiplicativity in the first variable, we immediately deduce the quadratic reciprocity law by taking $a,b$ to be odd primes.

An advantage of this formulation is that it generalizes to number fields $K$, by considering the completions of $K$ at all of its (finite and infinite) ``places''. Hilbert then went on to pose his 9th problem, in which he asked for a generalization of this law to $n$-th powers in arbitrary number fields; this is what class field theory achieves. We first consider the case of extensions which contain a primitive $n$-th root of unity:

\subsection{Kummer case}
Assume now $K|\Q$ is a finite extension such that $\mu_n\subset K$, and let $\OO_K$ be its ring of integers. If $P\subset \OO_K$ is a prime ideal, we would like to define a symbol $(a,b)_P\in \mu_n$ such that a product formula holds, i.e.
\[
	\prod_{P} (a,b)_P = 1
\]
if $a,b\in K^\times$. Moreover, for $P$ finite (which is always the case for $n\ne 2$) which does not divide $(n)$ we shall have \[
	(a,b)_P = \overline{(-1)^{v_P(a)v_P(b)}a^{v_P(b)}b^{-v_P(a)}}^{(q-1)/n},\] where $q = \#(\OO_K/P)$. In particular, if $P$ does not divide $(a)$, then $(a,b)_P =
	a^{\frac{q-1}{n}v_P(b)}\bmod P$.

So, if $v_P(b)=1$, then
\[
	\legendre{a}{P} \coloneqq (a,b)_P
\]
does not depend on the choice of $b$ and it is the unique $n$-th root of unity, that is $a^{\frac{q-1}{n}}\bmod P$; in other words, $\legendre{a}{P} = 1$ if and only if $x^n =a \bmod P$ has a solution. Some more properties of this symbol are:
\begin{enumerate}
	\item $(a,b)_P=1 \iff$ $b$ is a norm from $K_P(\sqrt[n]{a}) \iff$ $a$ is a norm from $K_P(\sqrt[n]{b})$;
	\item $(a,b)_P$ is multiplicative in both variables and anticommutative;
\end{enumerate}

When $\mu_n$ is \textit{not} contained in $K_P$, an ``Hilbert'' symbol still exists but is longer associated to $a,b\in K^\times$. Suppose for a moment that $\mu_n\subset K_P$. Then, the extension $K_P(\sqrt[n]{a})|K_P$ is Galois for all $a\notin K_P^{\times n}$, and its Galois group is
\[
	G_a\coloneqq\Gal(K_P(\sqrt[n]{a})|K_P) \cong \Z/n\Z
\]
Then, the group $ \Hom(G_a,\mu_n)\cong \Z/n\Z$ is generated by the character
\[
	\overline{\chi_a}:\sigma\mapsto \frac{\sigma(\sqrt[n]{a})}{\sqrt[n]{a}}\in \mu_n \cong \Z/n\Z.
\]
In general, we shall construct a generalized Hilbert symbol
\begin{align*}
	\Hom\left(\Gal(\overline{K_P}|K_P),\Z/n\Z\right) \times K_P^\times & \to \Z/n\Z,        \\
	[\chi,b]                                                           & \mapsto (\chi,b)_P
\end{align*}

and we'll thus be able to prove the \textbf{Artin} reciprocity law: fix some $b\in K^\times$, and suppose that $\chi$ is a character in $\Hom(G_K,\Z/n\Z)$. Then
\[
	\sum_P (\chi_P,b)=0,
\]
where $\chi_P$ is the restriction of $\chi$ to $G_{K_P}^\ab$ (recall that it embeds into $G_K^\ab$).

The generalized Hilbert symbol defines a reciprocity map
\[
	K_P^\times \to G_{K_P}^\ab,
\]
which is ``almost'' an isomorphism. In particular, for every $n$ we'll have an isomorphism of finite groups
\[
	K_P^\times/n \to G_{K_P}^\ab/n.
\]
This is one of the remarkable features of class field theory, giving a description of the abelian extensions of a field in terms of its arithmetic.
To define such a map, it's enough to define the dual map
\begin{align*}
	\Hom(G_{K_P}^\ab/n,\Z/n\Z) & \to \Hom(K_P^\times/n,\Z/n\Z), \\
	[\chi]                     & \mapsto [b\mapsto (\chi,b)_P].
\end{align*}

\section{Computing the Hilbert symbol}

Recall the definition: for $a,b\in\Q_p^\times$,
\[
	(a,b)_p = \begin{cases}
		1  & \text{if $ax^2+by^2=1$ is solvable in $\Q_p$}, \\
		-1 & \text{otherwise}.
	\end{cases}
\]
We have seen that
\[
	(a,b)_p = 1 \iff z^2 - ax^2 - by^2 \text{ has a non-trivial solution in } \Q_p^3 \iff b \text{ is a norm from } \Q_p(\sqrt{a}),
\]
where $N(x+y\sqrt{a}) = x^2 - ay^2$.

\begin{thm}[Hilbert reciprocity law]\label{thm:hilbertrecip}
	For all $a,b\in\Q^\times$,
	\[
		\prod_{p} (a,b)_p = 1,
	\]
	where the product runs over all primes $p$, including $p=\infty$.
\end{thm}
Note that the product is finite: $(a,b)_p=1$ for all $p\nmid 2ab$ (see Lemma~\ref{lem:units} below).

\begin{lem}\label{lem:basic}
	Let $a,b,c\in\Q_p^\times$. Then
	\begin{enumerate}
		\item $(1,c)_p = 1$;
		\item $(-c,c)_p = 1$;
		\item if $(a,c)_p = 1$, then $(ab,c)_p = (a,c)_p(b,c)_p$ (this holds in general, see the Corollary below);
		\item $(c,c)_p = (-1,c)_p$.
	\end{enumerate}
\end{lem}
\begin{proof}
	\begin{enumerate}
		\item Consider $N:\Q_p(\sqrt{c})\to\Q_p$. Then $N(1)=1$.
		\item If $c\in\Q_p^{\times 2}$, then $\Q_p(\sqrt{c})=\Q_p$, the norm is the identity and $N(-c)=-c$. If $c\notin\Q_p^{\times 2}$, then $N(\sqrt{c}) = -c$. In both cases $-c$ is a norm from $\Q_p(\sqrt{c})$.
		\item If $a=N(x)$ and $b=N(y)$, then $ab=N(x)N(y)=N(xy)$, since the norm is multiplicative by definition.

		      Moreover, if $a=N(x)$, then $a^{-1}=N(x^{-1})$; if in addition $ab=N(z)$, then
		      \[
			      b=a^{-1}(ab)=N(x^{-1})N(z)=N(x^{-1}z).
		      \]
		      So, if $(a,c)_p=1$ (i.e.\ $a=N(x)$ is a norm from $\Q_p(\sqrt{c})$), then $(b,c)_p=1\iff(ab,c)_p=1$.
		\item By (2) and (3),
		      \[
			      (c,c)_p = (-c,c)_p(c,c)_p = (-c^2,c)_p = (-1,c)_p,
		      \]
		      because in general $(ad^2,b)_p = (a,b)_p$.\qedhere
	\end{enumerate}
\end{proof}
\begin{rmk}
	Since $(ad^2,b)_p=(a,b)_p$, the Hilbert symbol is a well-defined map
	\[
		(\,\cdot\,,\cdot\,)_p : \left(\Q_p^\times/\Q_p^{\times 2}\right)\times\left(\Q_p^\times/\Q_p^{\times 2}\right)\to\mu_2.
	\]
\end{rmk}

\begin{lem}\label{lem:units}
	Let $p>2$. Then $(u,v)_p=1$ for all $u,v\in\Z_p^\times$.
	(Recall that $\Z_p$ is local and $\Z_p^\times=\Z_p\setminus(p)$.)
\end{lem}
\begin{proof}
	Let $\overline{u}\coloneqq u\bmod p$ and $\overline{v}\coloneqq v\bmod p$; they are non-zero elements of $\F_p$ (units of the local ring).
	We first show that $1-\overline{u}x^2-\overline{v}y^2$ has a solution in $\F_p$. Since $\F_p^\times$ is cyclic of order $p-1$,
	\[
		\#\{x^2 : x\in\F_p\} = \frac{p+1}{2}.
	\]
	Similarly, if $S_{\overline{u}}\coloneqq\{\overline{u}x^2:x\in\F_p\}$ and $S_{\overline{v}}\coloneqq\{1-\overline{v}y^2 : y\in\F_p\}$, then $\#S_{\overline{u}}=\#S_{\overline{v}}=
		\frac{p+1}{2}$.
	Thus, $S_{\overline u}\cap S_{\overline v}\neq\emptyset$ by the

	We conclude by Hensel's lemma:
\end{proof}

\begin{lem}[Hensel]
	Let $f\in\Z_p[X]$ and $a_0\in\Z_p$ with $f(a_0)\equiv 0\mod p$ and $f'(a_0)\not\equiv 0\mod p$. Then there exists $a\in\Z_p$ with $f(a)=0$ and $a\equiv a_0\mod p$.
\end{lem}
This allows us to lift a solution mod $p$ to a solution in $\Z_p$ (to apply it in the proof above, we have to fix one variable).

\begin{prop}\label{prop:oddp}
	Let $p>2$ and $a,b\in\Q_p^\times$, with $a=p^\alpha u$, $b=p^\beta v$ and $u,v\in\Z_p^\times$. Then
	\[
		(a,b)_p = (-1)^{\alpha\beta\frac{p-1}{2}}\legendre{\overline{u}}{p}^{-\beta}\legendre{\overline{v}}{p}^{\alpha}.
	\]
\end{prop}
\begin{proof}
	Since $(a,b)_p$ depends only on $a,b \bmod \Q_p^{\times 2}$, we may assume $\alpha,\beta\in\{0,1\}$.
	\begin{itemize}
		\item $\alpha=\beta=0$: this is Lemma~\ref{lem:units}.
		\item $\alpha=1$, $\beta=0$: we need $(pu,v)_p=\legendre{\overline{v}}{p}$. By Lemma~\ref{lem:units} we have $(u^{-1},v)_p=1$, hence by Lemma~\ref{lem:basic}(3)
		      \[
			      (pu,v)_p = (pu,v)_p\underbrace{(u^{-1},v)_p}_{=1} = (p,v)_p.
		      \]
		      If $v\in\Q_p^{\times 2}$, then $(p,v)_p=(p,1)_p=1=\legendre{\overline{v}}{p}$ (note that $\overline{v}$ is then also a square mod $p$).
		      If $v\notin\Q_p^{\times 2}$, then $(p,v)_p=-1$, because $z^2-px^2-vy^2$ has no non-trivial solution mod $p$, hence none in $\Q_p$ by homogeneity; also $\legendre{\overline{v}}{p}=-1$ (by Hensel, since $v\notin\Q_p^{\times 2}$).
		\item $\alpha=\beta=1$: we need $(pu,pv)_p=(-1)^{\frac{p-1}{2}}\legendre{\overline{u}}{p}\legendre{\overline{v}}{p}$. By Lemma~\ref{lem:basic}(2) we have $(-pv,pv)_p=1$, so by Lemma~\ref{lem:basic}(3)
		      \[
			      (pu,pv)_p = (pu,pv)_p\underbrace{(-pv,pv)_p}_{=1} = (-p^2uv,pv)_p = (-uv,pv)_p = (pv,-uv)_p,
		      \]
		      and now we are in the previous case, which gives
		      \[
			      (pv,-uv)_p = \legendre{\overline{-uv}}{p} = \legendre{-1}{p}\legendre{\overline{u}}{p}\legendre{\overline{v}}{p},
		      \]
		      where $\legendre{-1}{p}=(-1)^{\frac{p-1}{2}}$, and we used that the Legendre symbol is multiplicative.\qedhere
	\end{itemize}
\end{proof}

\begin{cor}
	Let $p>2$. Then $(a,c)_p(b,c)_p=(ab,c)_p$ for all $a,b,c\in\Q_p^\times$. In particular, $(\,\cdot\,,\cdot\,)_p$ is bi-multiplicative.
\end{cor}
\begin{proof}
	Write $a=p^\alpha u$, $b=p^\beta v$ and $c=p^\gamma w$ with $u,v,w\in\Z_p^\times$. By Proposition~\ref{prop:oddp},
	\begin{align*}
		(a,c)_p(b,c)_p
		 & = (-1)^{\frac{p-1}{2}\alpha\gamma}\legendre{\overline{u}}{p}^{\gamma}\legendre{\overline{w}}{p}^{\alpha}\cdot(-1)^{\frac{p-1}{2}\beta\gamma}\legendre{\overline{v}}{p}^{\gamma}\legendre{\overline{w}}{p}^{\beta} \\
		 & = (-1)^{\frac{p-1}{2}\gamma(\alpha+\beta)}\legendre{\overline{uv}}{p}^{\gamma}\legendre{\overline{w}}{p}^{\alpha+\beta} = (ab,c)_p.\qedhere
	\end{align*}
\end{proof}

\paragraph{Case $p=2$.}
If $a=2^\alpha u$ and $b=2^\beta v$ with $u,v\in\Z_2^\times$, define
\[
	\varepsilon(u)\coloneqq\frac{u-1}{2}\bmod 2,\qquad \omega(u)\coloneqq\frac{u^2-1}{8}\bmod 2.
\]
Then
\[
	(a,b)_2 = (-1)^{\varepsilon(u)\varepsilon(v)+\alpha\omega(v)+\beta\omega(u)}.
\]
(Proof not given.) In particular, $(\,\cdot\,,\cdot\,)_p$ is bi-multiplicative for every $p$.
\begin{proof}[Proof of Theorem \ref{thm:hilbertrecip}]
	Using bi-multiplicativity, we reduce to the case where each of $a,b$ is either $-1$ or a prime.
	\begin{itemize}
		\item Case $a=-1$: homework.
		\item Case $a=b=q$ prime: $(q,q)_p=(-1,q)_p$ by Lemma~\ref{lem:basic}(4), so we are reduced to the previous case.
		\item Case $a=2$, $b>2$ prime: $(2,b)_p=1$ for $p\notin\{2,b\}$ (also for $p=\infty$, as $2,b>0$). Moreover, by the $p=2$ formula,
		      \[
			      (2,b)_2=(-1)^{\omega(b)},
		      \]
		      and by Proposition~\ref{prop:oddp}
		      \[
			      (2,b)_b=\legendre{2}{b}=(-1)^{\frac{b^2-1}{8}}.
		      \]
		      The product of these two factors is $1$.
		\item Case $a\neq b$ odd primes: $(a,b)_p=1$ for $p\notin\{2,a,b\}$ (as well as $p=\infty$), and
		      \[
			      (a,b)_p = \begin{cases}
				      (-1)^{\frac{a-1}{2}\frac{b-1}{2}} & p=2, \\[2pt]
				      \legendre{a}{b}                   & p=b, \\[2pt]
				      \legendre{b}{a}                   & p=a.
			      \end{cases}
		      \]
		      Hence the product is $1$ by quadratic reciprocity.\qedhere
	\end{itemize}
\end{proof}

\section{Cohomology of groups}
Already for the local part of theory we'll require some input from homological algebra, mainly the description of the first cohomology group $H^1(G,A)$ for $G$ some Galois group, and its interpretation in terms of crossed homomorphisms. A concise reference for this material can be found in the notes \cite{homalg}. We recall some results here for the reader's convenience:

\begin{thm}[Cohomology of cyclic groups]
	Let $G$ be a finite cyclic group of order $n$, generated by $\sigma$. Then,
	\[
		\begin{cases}
			H^0(G,A) = A^G,           \\[2pt]
			H^1(G,A) = A/(1-\sigma)A, \\[2pt]
			H^2(G,A) = A^G/NA,        \\[2pt]
			H^i(G,A) \cong H^{i+2}(G,A) \text{ for all } i\ge 1,
		\end{cases}
	\]
\end{thm}
This comes out of the ``periodic'' resolution of $\Z$ as a trivial $G$-module:
\[
	\ldots \xrightarrow{\sigma-1} \Z[G] \xrightarrow{N} \Z[G] \xrightarrow{\sigma-1} \Z[G] \xrightarrow{N} \Z[G] \xrightarrow{\varepsilon} \Z \to 0,
\]
where $N = 1+\sigma+\ldots+\sigma^{n-1}$ is the norm map and $\varepsilon$ sends $\sigma$ to $1$.
A similar computation holds for the infinite cyclic group $G=\Z$:
\[
	H^0(\Z,A) = A^\Z,\qquad H^1(\Z,A) = A/(1-\sigma)A,\qquad H^i(\Z,A) = 0 \text{ for all } i\ge 2.
\]
For a subgroup $H\subset G$, we denote by
\[M^G_H(A)\coloneqq  \Hom_H(\Z[G],A)\] the $G$-module with action
\[
	\sigma(\phi)(g) = \phi(\sigma^{-1}g),\qquad \sigma\in G, g\in G, \phi\in M^G_H(A).
\]
Then, \textit{Shapiro}'s lemma tells us that there is a canonical isomorphism
\begin{equation}\label{eq:shapiro}
	H^i(G,M^G_H(A)) \cong H^i(H,A).
\end{equation}

\subsection{Restriction, Inflation and Corestriction}

Let $H$ be a subgroup of $G$. The \emph{restriction} map is defined on cocycles as
\[
	\operatorname{Res}: H^i(G,A) \to H^i(H,A),\quad \operatorname{Res}([f]) = [f|_H],
\]
which explains the name: $f|_H$ is just restriction to $H$.

For a normal subgroup $H\triangleleft G$, the quotient $G/H$ acts on $A^H$ and we have the \emph{inflation} map
\[
	\operatorname{Inf}: H^i(G/H,A^H) \to H^i(G,A),\quad \operatorname{Inf}([f]) = [f\circ \pi],
\]
where $\pi:G\to G/H$ is the canonical projection.

\subsection{Profinite groups}

Let $G=\varprojlim G_\alpha$ be a profinite group (that is, an inverse limit of finite groups). This is a topological group, whose topology is induced by the discrete topology on the finite groups $G_\alpha$. 

We have natural continuous maps $\pi_\alpha:G\to G_\alpha$, and their kernels $U_\alpha$ form a system of open neighborhoods of $1\in G$.

\begin{rmk}\
	\begin{itemize}
			\item We may assume that the $G_\alpha$ coincide with \textbf{all} finite quotients of $G$ by open normal subgroups, as if $\Gamma$ is such a quotient, then there is some $\alpha$ such that $U_\alpha$ has trivial image in $\Gamma$: indeed, $\Gamma$ is finite and the intersection of all $U_\alpha$ is trivial. So, we get a factorization
		\[
		\begin{tikzcd}
			G \arrow[rd, "\pi_\alpha"] \arrow[d] &   \\
			G_\alpha \arrow[r, dashed]                     & \Gamma
		\end{tikzcd}
		\]
		\item The inverse limit of a system of finite groups is compact and totally disconnected. With compact here we also include the Hausdorff property, which holds by Tychonoff's theorem. Moreover, the open subgroups of $G$ are exactly the closed subgroups of finite index.
	\end{itemize}	
\end{rmk}
\subsection{Cohomology of profinite groups}
Let $G=\varprojlim G_\alpha$ be a profinite group, and let $A$ be a \textit{continuous} $G$-module, i.e. an abelian group with an action of $G$ by automorphisms such that $\forall a\in A$, the stabilizer $G_a$ is open in $G$ (exercise: this is equivalent to the map $G\times A\to A$ being continuous, where $A$ has the discrete topology). In particular, if $G_\alpha=G/U_\alpha$, then $A^{U_\alpha}$ is a $G_\alpha$-module, and we define:
\begin{defn}
	\[
		H^i_{\text{cont}}(G,A) \coloneqq \varinjlim H^i(G_\alpha,A^{U_\alpha}),
	\]
	where the limit is taken with respect to the inflation maps induced by the projections $G_\beta\twoheadrightarrow G_\alpha$:
	\[
		H^i(G_\alpha,A^{U_\alpha}) \to H^i(G_\beta,A^{U_\beta}).
	\]
	We'll immediately drop the subscript $\text{cont}$, writing $H^i(G,A)$.
\end{defn}
In the case where $G=\Gal(k^s|k)$ is the absolute Galois group of a field $k$, we'll use the notation $H^i(k,A)$ for $H^i(G,A)$.

\begin{prop}
		If $G$ is profinite and $A$ is a continuous $G$-module, then $H^i(G,A)$ is a torsion abelian group for all $i>0$.
\end{prop}
\begin{proof}
		The groups $H^i(G_\alpha,A^{U_\alpha})$ are torsion: they are killed by the order of $G_\alpha$. Therefore, their direct limit $H^i(G,A)$ is also torsion.
\end{proof}
\begin{cor}
		If moreover $A$ is a $\Q$-vector space, then $H^i(G,A)=0$ for all $i>0$.
\end{cor}
\begin{rmk}
		Given a finite group $G$ and a finitely generated $G$-module $A$, the cohomology groups $H^i(G,A)$ are finite for all $i>0$. This follows from the description of cohomology in terms of the standard resolution and the torsion property above.
\end{rmk}

\begin{lem}
		Let $G$ be profinite with $(U_\alpha)$ a system of open normal subgroups. Then,
		\begin{itemize}
			\item If $H$ is a closed subgroup of $G$, there is a canonical isomorphism
			\[
				H^i(H,A) \cong \varinjlim H^i(H/(H\cap U_\alpha),A^{H\cap U_\alpha})
			\]
			which shows that $H$ is profinite;
			\item If moreover $H$ is normal in $G$, then $G/H\cong \varprojlim G/(HU_\alpha)$ is profinite.
		\end{itemize}
	\end{lem}
Therefore, we can define restriction, corestriction and inflation maps for profinite groups by taking the direct limit of the corresponding maps for finite groups: for instance, restriction is induced by the maps
\[
	H^i(G/U_\alpha,A^{U_\alpha}) \xrightarrow{\operatorname{res}} H^i(H/(H\cap U_\alpha),A^{H\cap U_\alpha}).
\]

Since we defined cohomology explicitly as a direct limit, we have to show ``by hand'' the existence of the long exact sequence in cohomology associated with a short exact sequence of $G$-modules $0\to A\to B\to C\to 0$.

\begin{lem}
		Let $(A_\alpha,\phi_{\alpha\beta})$, $(B_\alpha,\psi_{\alpha\beta})$ and $(C_\alpha,\rho_{\alpha\beta})$ be direct systems of $G_\alpha$-modules, and assume given short exact sequences
		\[
			0 \to A_\alpha \xrightarrow{f_\alpha} B_\alpha \xrightarrow{g_\alpha} C_\alpha \to 0
		\]
		for all $\alpha$ which are compatible with the transition maps (i.e. morphisms of direct systems). Then, the direct limit of these sequences is exact.
\end{lem}

The non-trivial part is defining the connecting homomorphism
\[
	\delta: H^i(G,C) \to H^{i+1}(G,A).
\]
Let $U\subset G$ be an open normal subgroup. If $K_U\coloneqq \coker(B^U\to C^U,)$ then $\varinjlim_U K_U=0$. By the lemma, we have a short exact sequence
\[
	\varinjlim H^i(G/U,(B^U/A^U)) \to \varinjlim H^i(G/U,C^U) \to \varinjlim H^{i+1}(G/U,K_U) 
\]
and the last term is $0$ by the above. Therefore, for small enough $U$ we can lift a cocycle $f\in Z^i(G,C)$ to some element in $\varinjlim H^i(G /U,(B^U/A^U))$, and then apply the connecting homomorphism coming from the short exact sequence $0\to A^U \to B^U\to B^U/A^U\to 0$ to get an element in $\varinjlim H^{i+1}(G/U,A^U)$, which has a map into $H^{i+1}(G,A)$ (of course, one should check that this does not depend on any of the choices made).

\section{Kummer Theory}
\begin{prop}[Hilbert's Theorem 90]
	Let $K|k$ be a finite Galois extension with group $G=\Gal(K|k)$. Then
	\[
		H^1(G,K^\times) = 0.
	\]
	Consequently, by passing to the limit over all such extensions we get
	\[
	 	H^1(k,(k^s)^\times) = 0.
	\]
		
\end{prop}
\begin{lem}[Dedekind]
	If $\sigma_1,\ldots,\sigma_n$ are the elements of $G$, than they are linearly independent over $K$.
\end{lem}
\begin{proof}
		Suppose they are linearly dependent, and let
		\[
			a_1\sigma_1+\ldots+a_n\sigma_n=0
		\]
		be the \textit{shortest} nontrivial linear relation. Then, choosing $x\in K$ such that $\sigma_1(x)\ne \sigma_2(x)$ and $y\in K$ is arbitrary, then
		\begin{align*}
			0 & = a_1\sigma_1(xy)+\ldots+a_n\sigma_n(xy) \\
			  & = a_1\sigma_1(x)\sigma_1(y)+\ldots+a_n\sigma_n(x)\sigma_n(y) \\
			  & = \sigma_1(x)(a_1\sigma_1(y)+\ldots+a_n\sigma_n(y)) + (\sigma_2(x)-\sigma_1(x))(a_2\sigma_2(y)+\ldots+a_n\sigma_n(y)).
		\end{align*}
		Putting this together with the original relation evaluated at $y$ and multiplied by $\sigma_1(x)$, we get a shorter relation
		\[
			(\sigma_2(x)-\sigma_1(x))(a_2\sigma_2(y)+\ldots+a_n\sigma_n(y)) = 0,
		\]
		which is a contradiction.
\end{proof}

		Let $\varphi:G\to K^\times$ be a $1$-cocycle, with the cocycle condition being $\varphi(\sigma\tau) = \varphi(\sigma)\cdot\sigma(\varphi(\tau))$. We want to show that $\varphi$ is a coboundary, i.e. there exists some $y\in K^\times$ such that $\varphi(\tau) = \tau(y)\cdot y^{-1}$ for all $\tau\in G$. Consider the element
		\[
			\Phi = \sum_{\sigma\in G} \varphi(\sigma)\cdot\sigma \in K[G].
		\]
		By the lemma, $\Phi$ is non-zero. Therefore there exist $x,y\in K^\times$ such that
		\[
			y= \Phi(x)= \sum_{\sigma\in G} \varphi(\sigma)\cdot\sigma(x),
		\]
		and applying $\tau$ to both sides gives
		\[
			\tau(y) = \sum_{\sigma\in G} \tau \varphi(\sigma)\cdot\tau\sigma(x) = \sum_{\sigma\in G} \frac{\varphi(\tau\sigma)}{\varphi(\tau)}\tau\sigma(x)= \frac{1}{\varphi(\tau)}\sum_{\sigma\in G} \varphi(\tau\sigma)\cdot\tau\sigma(x) = \frac{1}{\varphi(\tau)}\sum_{\sigma\in G} \Phi(x)= \frac{y}{\varphi(\tau)},
		\]
\end{proof}
The main result of this section is the following:
\begin{prop}
		Let $k$ be a field and $m$ a positive integer which is prime to $\operatorname{char}(k)$ (this implies that $\mu_m\subset k^s$). Then, there is a canonical isomorphism
		\[
			H^1(k,\mu_m) \cong k^\times/k^{\times m}.
		\]	
		whose inverse sends the class of $a\in k^\times/k^{\times m}$ to the class of the cocycle $\sigma \mapsto \sigma(\alpha)\cdot \alpha^{-1}$, where $\alpha\in (k^s)^\times$ is any $m$-th root of $a$.
\end{prop}
\begin{proof}
		Consider the short exact sequence of $G_k$-modules
		\[
			1 \to \mu_m \to (k^s)^\times \xrightarrow{(\cdot)^m} (k^s)^\times \to 1.
		\]
		By Hilbert's Theorem 90, we have $H^1(k,(k^s)^\times)=0$. Therefore, the long exact sequence in cohomology gives
		\[
			H^0(k,(k^s)^\times) \xrightarrow{(\cdot)^m} H^0(k,(k^s)^\times) \to H^1(k,\mu_m) \to 0,
		\]
		and the cokernel of the first map is exactly $k^\times/k^{\times m}$, since $H^0(k,(k^s)^\times) = (k^s)^{\times G_k} = k^\times$. The last statement then follows by writing out the connecting map explicitly (lifting corresponds to taking an $m$-th root\ldots).
\end{proof}
\begin{exmp}
		If moreover $\mu_m\subset k$, then $H^1(k,\mu_m) = H^1(k,\Z/m\Z)=\Hom(G_k,\Z/m\Z)$ (the action is trivial). In this case, we deduce that every finite extension $K|k$ with $\Gal(K|k)\cong \Z/m\Z$ is of the form $K=k(\alpha)$ with $\alpha^m\in k$.
\end{exmp}
\section{Cup-products}
Let $G$ be a group and $A,B$ be $G$-modules. The module $A\otimes_\Z B$ is a $G$-module with the diagonal action $\sigma(a\otimes b) = \sigma(a)\otimes\sigma(b)$. The cup-product is a bilinear pairing
\begin{align*}
	\cup: H^i(G,A)\times H^j(G,B) &\to H^{i+j}(G,A\otimes B)\\
	(a,b) &\mapsto a\cup b
\end{align*}
which is functorial in $A,B$ and graded-commutative: $a\cup b = (-1)^{ij}b\cup a$; there will be a version for profinite groups by passing to the limit. Let us sketch the application of this gadget to the Hilbert symbol:
\begin{defn}
		For a field $k$, the \emph{Brauer group} of $k$ is 
		\[
			\Br(k)\coloneqq H^2(k,k_s^\times).
		\]
		A non-trivial result is that elements of $\Br(k)$ correspond to central division algebras which are finite-dimensional over $k$.
\end{defn}
Another piece of the Kummer long exact sequence is
\[
	H^1(k,k_s^\times)=0 \to H^2(k,\mu_n) \xrightarrow{(\cdot)^n} H^2(k,k_s^\times),
\]
from which we get $H^2(k,\mu_n)\cong {_n\Br(k)}$, the $n$-torsion in the Brauer group.

Suppose that $\mu_n\subset k$, and take $a,b\in k^\times$. Then, we can take their classes in $k^\times/k^{\times n}\cong H^1(k,\mu_n)=H^1(k,\Z/n\Z)$, and take their cup-product:
\[
	H^1(k,\Z/n\Z)\times H^1(k,\Z/n\Z) \to H^2(k,\Z/n\Z\otimes_\Z \Z/n\Z) \cong {_n\Br(k)}.
\]
\begin{thm}[Hasse]
		Let $k|\Qp$ be a finite extension. Then,
		\[
			\Br(k) \cong \Q/\Z.	
		\]
\end{thm}
In particular, we deduce that for any $n$ ${_n\Br(k)}\cong \frac{1}{n}\Z/\Z$. Putting these together, we get a map
\[
	(\cdot,\cdot)_n: k^\times/k^{\times n}\times k^\times/k^{\times n} \to H^1(k,\Z/n\Z)\times H^1(k,\Z/n\Z) \to {_n\Br(k)} \cong \Z/n\Z,
\]
which is the \emph{Hilbert symbol}.
What happens if $\mu_n\not\subset k$? Given a character $\chi:G_k\to\Z/n\Z$, this lives in the first cohomology group $H^1(k,\Z/n\Z)$, so we can take its cup-product with the class of any $b\in k^\times$ inside $H^1(k,\mu_n)$:
\[
	(\chi,b)\coloneqq \chi\cup b \in H^2(k,\mu_n) \overset{\text{Hasse}}{\cong} \Z/n\Z.
\]

\begin{constr}
		Let $A^\bullet,B^\bullet$ be complexes of e.g. abelian groups. The tensor product of these complexes is defined as follows: consider the double complex
		\[
			\begin{tikzcd}
				\vdots & \vdots & \vdots & \\
				A^{i-1}\otimes B^{j+1} \arrow[r] \arrow[u] & A^{i}\otimes B^{j+1} \arrow[r] \arrow[u] & A^{i+1}\otimes B^{j+1} \arrow[r] \arrow[u] & \ldots \\
				A^{i-1}\otimes B^{j} \arrow[r] \arrow[u] & A^{i}\otimes B^{j} \arrow[r] \arrow[u] & A^{i+1}\otimes B^{j} \arrow[r] \arrow[u] & \ldots \\
				A^{i-1}\otimes B^{j-1} \arrow[r] \arrow[u] & A^{i}\otimes B^{j-1} \arrow[r] \arrow[u] & A^{i+1}\otimes B^{j-1} \arrow[r] \arrow[u] & \ldots
			\end{tikzcd}
		\]
		with horizontal and vertical differentials
		\[
			\partial^h_{i,j} = \partial_A^i\otimes\id_{B^j},\qquad \partial^v_{i,j} = (-1)^i\id_{A^i}\otimes\partial_B^j.
		\]
		With this sign, the diagram \text{anti-}commutes, i.e. $\partial^h_{i,j+1}\circ\partial^v_{i,j} =-\partial^v_{i+1,j}\circ\partial^h_{i,j}$. Recall that the total complex $T(A^\bullet\otimes B^\bullet)$ in this case is defined as
		\[
			T^n(A^\bullet\otimes B^\bullet) = \bigoplus_{i+j=n} A^i\otimes B^j,	
		\]
		with differential given on each component by the sum of the horizontal and vertical differentials:
		\[
			\partial^n = \sum_{i+j=n} \partial^h_{i,j} + \partial^v_{i,j}.
		\]
		Because of the anti-commutativity, we have $\partial^{n+1}\circ\partial^n=0$, which shows that this is a complex.
		\begin{defn}
				We denote by $(A\otimes B)^\bullet$ the total complex $T(A^\bullet\otimes B^\bullet)$, and call it the \emph{tensor product} of the two complexes.
		\end{defn}
		Assume moreover that $A,B$ are abelian groups, and consider the complexes $\Hom(A^\bullet,A)$ and $\Hom(B^\bullet,B)$, with differentials given by pre-composition with $\partial_A,\partial_B$. We want to define a product
		\[
				H^i(\Hom(A^\bullet,A))\times H^j(\Hom(B^\bullet,B)) \to H^{i+j}(\Hom((A\otimes B)^\bullet,A\otimes B)).
				\]
		If $\alpha\in \Hom(A^i,A)$ and $\beta\in \Hom(B^j,B)$, then $\alpha\otimes\beta: A^{-i}\otimes B^{-j}\to A\otimes B$ and we get a map
		\[
				\Hom(A^{-i}\otimes B^{-j},A\otimes B) \to \Hom\left(\bigoplus_{k+l=i+j} A^{-k}\otimes B^{-l},A\otimes B\right) = \Hom(T^{i+j}(A^\bullet\otimes B^\bullet),A\otimes B),
		\]
		and one easily checks that this is compatible with differentials (sends cocycles to cocycles and coboundaries to coboundaries). Therefore, it induces such a product on cohomology, which is the desired cup-product.
		\begin{lem}
				Let $P_\bullet \to \Z$ be a projective resolution of $\Z$ as a trivial $G$-module, and recall the homological indexing $P_i=P^{-i}$. Then, $(P\otimes P)_\bullet$ is a projective resolution of $\Z$ as a trivial $(G\times G)$-module, where the action is the diagonal one.
		\end{lem}
		Applying the previous construction with $A^\bullet =B^\bullet = P_\bullet$, we get a product
		\[
			H^i(\Hom_G(P_\bullet,A))\times H^j(\Hom_G(P_\bullet,B)) \to H^{i+j}(\Hom_{G\times G}((P\otimes P)_\bullet,A\otimes B)),	
		\]
		and composing with the map $H^{i+j}(\Hom_{G\times G}((P\otimes P)_\bullet,A\otimes B)) \to H^{i+j}(\Hom_G((P\otimes P)_\bullet,A\otimes B))$ induced by the diagonal embedding $G\hookrightarrow G\times G$, we get a cup-product
		\[
			H^i(G,A)\times H^j(G,B) \to H^{i+j}(G,A\otimes B).
		\]
		This is compatible with restriction, inflation and corestriction maps: if $H\leq G$ is a subgroup, then
		\begin{itemize}
				\item For $a\in H^i(G,A)$ and $b\in H^j(G,B)$, we have $\operatorname{Res}(a\cup b) = \operatorname{Res}(a)\cup \operatorname{Res}(b) \in H^{i+j}(H,A\otimes B)$.
				\item If $H\triangleleft G$ is normal, then for $a\in H^i(G/H,A^H)$ and $b\in H^j(G/H,B^H)$, we have $\operatorname{Inf}(a\cup b) = \operatorname{Inf}(a)\cup \operatorname{Inf}(b) \in H^{i+j}(G,A\otimes B)$.
				\item If $[G:H]<\infty$, then for $a\in H^i(H,A)$ and $b\in H^j(G,B)$, we have 
				\[
					\operatorname{Cor}(a\cup \operatorname{Res}(b)) = \operatorname{Cor}(a)\cup b \in H^{i+j}(G,A\otimes B)
				\]
				which is called the \emph{projection formula}.
		\end{itemize}
\end{constr}
